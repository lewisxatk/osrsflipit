function fuzzyScore(query,text){
  const q=query.toLowerCase().trim().replace(/[^a-z0-9\s']/g," ").replace(/\s+/g," ");
  const t=text.toLowerCase().replace(/[^a-z0-9\s']/g," ").replace(/\s+/g," ").trim();
  if(!q)return 0;
  if(t===q)return 100000;
  const qt=q.split(" ").filter(Boolean), tt=t.split(" ").filter(Boolean);
  let score=0, matched=0;
  for(const token of qt){
    let best=0;
    for(const word of tt){
      if(word===token)best=Math.max(best,1000);
      else if(word.startsWith(token))best=Math.max(best,820-Math.max(0,word.length-token.length)*8);
      else if(word.includes(token))best=Math.max(best,650-Math.max(0,word.indexOf(token))*10);
      else {const d=levenshtein(token,word), maxLen=Math.max(token.length,word.length);if(d<=Math.max(1,Math.floor(maxLen*.34)))best=Math.max(best,560-d*65-Math.max(0,maxLen-token.length)*8)}
    }
    if(best>0){score+=best;matched++}
  }
  const all=qt.join(" ");
  if(t.includes(all))score+=1800;
  if(t.startsWith(all))score+=2200;
  const coverage=matched/qt.length;
  score+=coverage*900;
  if(matched===0||coverage<.5)return -1;
  return score-(t.length-q.length)*0.35;
}

async function api(path) {
  const key = `osrsflipit-cache:${path}`;
  const ttl = path.startsWith("/timeseries") ? 5 * 60 * 1000 : 45 * 1000;
  try {
    const cached = JSON.parse(sessionStorage.getItem(key) || "null");
    if (cached && Date.now() - cached.at < ttl) return cached.data;
  } catch {}
  const controller = new AbortController();
  const timeout = setTimeout(() => controller.abort(), path.startsWith("/timeseries") ? 8500 : 12000);
  try {
    const res = await fetch(`${API}${path}`, { headers: { Accept: "application/json" }, signal: controller.signal });
    if (!res.ok) throw new Error(`${res.status} ${res.statusText}`);
    const data = await res.json();
    try { sessionStorage.setItem(key, JSON.stringify({at:Date.now(),data})); } catch {}
    return data;
  } catch (e) {
    try {
      const cached = JSON.parse(sessionStorage.getItem(key) || "null");
      if (cached?.data) return cached.data;
    } catch {}
    throw e;
  } finally { clearTimeout(timeout); }
}

function scoreItem(x){

  const volumeScore=clamp((Math.log10(Math.max(0,x.volume)+1)/Math.log10(250000+1))*100);
  const capitalScore=clamp((x.buy>0?(x.margin/x.buy)*1000000/50000:0));
  const liquidityScore=clamp((x.volume/Math.max(1,Math.ceil(1000000/Math.max(1,x.buy))))*100);
  const riskScore=Math.round(clamp(volumeScore*.42+liquidityScore*.28+clamp(100-Math.max(0,x.roi-3)*7)*.18+clamp(100-(x.buy/50000000)*35)*.12));
  const penalty=x.buy>100000000?Math.min(30,Math.log10(x.buy/100000000+1)*30):0;
  const flipScore=Math.round(clamp(roiScore*.24+marginScore*.24+volumeScore*.22+riskScore*.18+capitalScore*.12-penalty));
  const capitalEfficiency=x.buy>0?(x.margin/x.buy)*1000000:0;
  const limitCost=x.buy>0&&x.limit>0?x.buy*x.limit:0;
  const slotQty=Math.max(1,Math.min(x.limit||1,Math.floor(1000000/Math.max(1,x.buy))));
  const slotProfit=x.margin>0?x.margin*slotQty:0;
  const riskLabel=riskScore>=75?"Low":riskScore>=50?"Medium":"High";
  return {...x,flipScore,riskScore,riskLabel,capitalEfficiency,slotProfit,limitCost,scoreBreakdown:{roi:Math.round(roiScore),margin:Math.round(marginScore),volume:Math.round(volumeScore),risk:riskScore,capital:Math.round(capitalScore)}};
}
