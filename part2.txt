function ColumnResizeEffect({resizingColumn,setResizingColumn,setColumnWidths}){useEffect(()=>{if(!resizingColumn)return;const move=e=>setColumnWidths(ws=>({...ws,[resizingColumn.key]:Math.max(70,resizingColumn.startWidth+e.clientX-resizingColumn.startX)}));const up=()=>setResizingColumn(null);window.addEventListener("pointermove",move);window.addEventListener("pointerup",up,{once:true});return()=>{window.removeEventListener("pointermove",move);window.removeEventListener("pointerup",up)}} ,[resizingColumn]);return null}
